package waitlist

import (
	"errors"
	"net/mail"
	"strings"
	"time"
)

var (
	ErrInvalidEmail         = errors.New("invalid email address")
	ErrRegistrationNotFound = errors.New("registration not found")
	ErrConfigNotFound       = errors.New("waitlist configuration not found")
	ErrInvalidConfig        = errors.New("invalid waitlist configuration")
)

type Registration struct {
	RegistrationID string `json:"registration_id"`
	Email          string `json:"-"`
	Position       int64  `json:"position"`

	LaunchUpdatesConsent bool `json:"-"`
	MarketingConsent     bool `json:"-"`

	ConsentRecordedAt time.Time `json:"-"`

	LaunchUpdatesUnsubscribedAt *time.Time `json:"-"`
	MarketingUnsubscribedAt     *time.Time `json:"-"`

	CreatedAt time.Time `json:"created_at"`
}

type RegistrationInput struct {
	Email string

	LaunchUpdatesConsent bool
	MarketingConsent     bool
}

type RegistrationResult struct {
	Registration    Registration
	IsExisting      bool
	TotalRegistered int64
}

type WaitlistConfig struct {
	ConfigID  string    `json:"config_id"`
	LaunchAt  time.Time `json:"launch_at"`
	Timezone  string    `json:"timezone"`
	UpdatedAt time.Time `json:"updated_at"`
	UpdatedBy *string   `json:"-"`
}

type WaitlistConfigInput struct {
	LaunchAt  time.Time
	Timezone  string
	UpdatedBy *string
}

// NormalizeEmail ensures email addresses are stored
// in a consistent lowercase, trimmed format.
func NormalizeEmail(email string) (string, error) {
	normalised := strings.ToLower(strings.TrimSpace(email))

	if normalised == "" || len(normalised) > 254 {
		return "", ErrInvalidEmail
	}

	parsed, err := mail.ParseAddress(normalised)
	if err != nil {
		return "", ErrInvalidEmail
	}

	// Reject display-name formats such as:
	// Edmond <edmond@example.com>
	if parsed.Address != normalised {
		return "", ErrInvalidEmail
	}

	return normalised, nil
}

// =====================================================
// JSON RESPONSE HELPERS
// =====================================================

func writeJSON(
	w http.ResponseWriter,
	status int,
	payload any,
) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)

	_ = json.NewEncoder(w).Encode(payload)
}

func writeError(
	w http.ResponseWriter,
	status int,
	message string,
) {
	writeJSON(
		w,
		status,
		errorResponse{
			Error: message,
		},
	)
}

// =====================================================
// JSON REQUEST VALIDATION
// =====================================================

// decodeJSON enforces:
// - application/json content type
// - maximum request body size of 16 KB
// - rejection of unknown JSON fields
// - exactly one JSON object per request
func decodeJSON(
	w http.ResponseWriter,
	r *http.Request,
	destination any,
) error {

	contentType := r.Header.Get("Content-Type")

	mediaType, _, err := mime.ParseMediaType(contentType)
	if err != nil || mediaType != "application/json" {
		return errors.New("content type must be application/json")
	}

	r.Body = http.MaxBytesReader(
		w,
		r.Body,
		16*1024,
	)

	decoder := json.NewDecoder(r.Body)

	decoder.DisallowUnknownFields()

	if err := decoder.Decode(destination); err != nil {
		return err
	}

	// Reject additional JSON values or trailing content.
	var extra any

	if err := decoder.Decode(&extra); err != io.EOF {
		if err == nil {
			return errors.New("request must contain exactly one JSON object")
		}

		return err
	}

	return nil
}

// handleJSONDecodeError provides consistent HTTP responses.
func handleJSONDecodeError(
	w http.ResponseWriter,
	err error,
) {
	var maxBytesError *http.MaxBytesError

	if errors.As(err, &maxBytesError) {
		writeError(
			w,
			http.StatusRequestEntityTooLarge,
			"Request body is too large.",
		)
		return
	}

	writeError(
		w,
		http.StatusBadRequest,
		"Invalid request body.",
	)
}

// =====================================================
// PUBLIC REGISTRATION
// POST /v1/waitlist/register
// =====================================================

func (h *Handler) Register(
	w http.ResponseWriter,
	r *http.Request,
) {
	var request registerRequest

	if err := decodeJSON(w, r, &request); err != nil {
		handleJSONDecodeError(w, err)
		return
	}

	result, err := h.service.Register(
		r.Context(),
		RegistrationInput{
			Email:                request.Email,
			LaunchUpdatesConsent: request.LaunchUpdatesConsent,
			MarketingConsent:     request.MarketingConsent,
		},
	)

	if err != nil {
		if errors.Is(err, ErrInvalidEmail) {
			writeError(
				w,
				http.StatusUnprocessableEntity,
				"Please provide a valid email address.",
			)
			return
		}

		// Do not expose database errors or internal details.
		writeError(
			w,
			http.StatusInternalServerError,
			"Registration could not be completed.",
		)
		return
	}

	status := http.StatusCreated

	message := "Your waitlist registration was successful."

	if result.IsExisting {
		status = http.StatusOK
		message = "This email is already registered."
	}

	writeJSON(
		w,
		status,
		registerResponse{
			Message:           message,
			Position:          result.Registration.Position,
			TotalRegistered:   result.TotalRegistered,
			AlreadyRegistered: result.IsExisting,
		},
	)
}

// =====================================================
// PUBLIC LAUNCH CONFIGURATION
// GET /v1/waitlist/config
// =====================================================

func (h *Handler) GetConfig(
	w http.ResponseWriter,
	r *http.Request,
) {
	info, err := h.service.GetLaunchInfo(r.Context())

	if err != nil {
		if errors.Is(err, ErrConfigNotFound) {
			writeError(
				w,
				http.StatusServiceUnavailable,
				"Launch information is not available yet.",
			)
			return
		}

		writeError(
			w,
			http.StatusInternalServerError,
			"Unable to retrieve launch information.",
		)
		return
	}

	writeJSON(
		w,
		http.StatusOK,
		configResponse{
			LaunchAt: info.LaunchAt,
			Timezone: info.Timezone,
			Status:   info.Status,
		},
	)
}

// =====================================================
// ADMIN LAUNCH CONFIGURATION
// PUT /v1/admin/waitlist/config
// =====================================================

func (h *Handler) UpdateConfig(
	w http.ResponseWriter,
	r *http.Request,
) {
	var request updateConfigRequest

	if err := decodeJSON(w, r, &request); err != nil {
		handleJSONDecodeError(w, err)
		return
	}

	// Administrator identity must come from authenticated
	// middleware, never from the incoming request body.
	adminID, ok := r.Context().
		Value(adminIDContextKey{}).(string)

	if !ok || !validUUID(adminID) {
		writeError(
			w,
			http.StatusUnauthorized,
			"Administrator authentication required.",
		)
		return
	}

	config, err := h.service.UpdateLaunchConfig(
		r.Context(),
		WaitlistConfigInput{
			LaunchAt:  request.LaunchAt,
			Timezone:  request.Timezone,
			UpdatedBy: &adminID,
		},
	)

	if err != nil {
		if errors.Is(err, ErrInvalidConfig) {
			writeError(
				w,
				http.StatusUnprocessableEntity,
				"Invalid launch date or timezone.",
			)
			return
		}

		writeError(
			w,
			http.StatusInternalServerError,
			"Unable to update launch configuration.",
		)
		return
	}

	// Retrieve the authoritative status from the service.
	// Do not assume every updated launch date is upcoming.
	launchInfo, err := h.service.GetLaunchInfo(r.Context())

	if err != nil {
		writeError(
			w,
			http.StatusInternalServerError,
			"Configuration was saved, but launch status could not be retrieved.",
		)
		return
	}

	writeJSON(
		w,
		http.StatusOK,
		configResponse{
			LaunchAt: config.LaunchAt,
			Timezone: config.Timezone,
			Status:   launchInfo.Status,
		},
	)
}

// =====================================================
// SECURE UNSUBSCRIBE
// POST /v1/waitlist/unsubscribe
// =====================================================

func (h *Handler) Unsubscribe(
	w http.ResponseWriter,
	r *http.Request,
) {
	if h.tokens == nil {
		writeError(
			w,
			http.StatusServiceUnavailable,
			"Unsubscribe service is unavailable.",
		)
		return
	}

	var request unsubscribeRequest

	if err := decodeJSON(w, r, &request); err != nil {
		handleJSONDecodeError(w, err)
		return
	}

	token := strings.TrimSpace(request.Token)

	if token == "" {
		writeError(
			w,
			http.StatusBadRequest,
			"Unsubscribe token is required.",
		)
		return
	}

	// Verify the signed token before accessing registration data.
	claims, err := h.tokens.Verify(token)

	if err != nil {
		writeError(
			w,
			http.StatusBadRequest,
			"Invalid or expired unsubscribe link.",
		)
		return
	}

	// Process only the communication channel authorised
	// by the signed token.
	switch claims.Channel {

	case UnsubscribeLaunchUpdates:

		_, err = h.service.UnsubscribeLaunchUpdatesByID(
			r.Context(),
			claims.RegistrationID,
		)

	case UnsubscribeMarketing:

		_, err = h.service.UnsubscribeMarketingByID(
			r.Context(),
			claims.RegistrationID,
		)

	default:

		writeError(
			w,
			http.StatusBadRequest,
			"Invalid unsubscribe request.",
		)
		return
	}

	if err != nil {
		if errors.Is(err, ErrRegistrationNotFound) {
			writeError(
				w,
				http.StatusNotFound,
				"Registration not found.",
			)
			return
		}

		writeError(
			w,
			http.StatusInternalServerError,
			"Unable to process unsubscribe request.",
		)
		return
	}

	writeJSON(
		w,
		http.StatusOK,
		unsubscribeResponse{
			Message: "Your communication preference has been updated.",
		},
	)
}

// validUUID checks whether a value has the standard UUID format.
func validUUID(value string) bool {
	if len(value) != 36 {
		return false
	}

	for i, character := range value {
		if i == 8 || i == 13 || i == 18 || i == 23 {
			if character != '-' {
				return false
			}
			continue
		}

		if !((character >= '0' && character <= '9') ||
			(character >= 'a' && character <= 'f') ||
			(character >= 'A' && character <= 'F')) {
			return false
		}
	}

	return true
}

package waitlist

import (
	"net/http"
	"time"

	"github.com/go-chi/chi/v5"
)

// =====================================================
// ROUTE CONFIGURATION
// =====================================================

type RouteConfig struct {
	AdminToken     string
	AdminID        string
	AllowedOrigins []string
}

// =====================================================
// ROUTE REGISTRATION
// =====================================================

// RegisterRoutes registers all waitlist API endpoints.
//
// CORS is intentionally configured separately so the main
// application can install it once at the router level.
func RegisterRoutes(
	router chi.Router,
	handler *Handler,
	config RouteConfig,
) {
	// Public registration rate limiter.
	//
	// Default:
	// 5 registration attempts per IP address per minute.
	registrationLimiter := NewRegistrationRateLimiter(
		5,
		time.Minute,
	)

	// -------------------------------------------------
	// PUBLIC WAITLIST ENDPOINTS
	// -------------------------------------------------

	router.Route("/v1/waitlist", func(r chi.Router) {

		// Register a resident.
		r.With(
			registrationLimiter.Middleware,
		).Post(
			"/register",
			handler.Register,
		)

		// Return the total number of registrations.
		r.Get(
			"/count",
			handler.Count,
		)

		// Return public launch information.
		r.Get(
			"/config",
			handler.GetConfig,
		)

		// Process a signed unsubscribe token.
		// POST prevents the token from being placed in
		// the URL query string.
		r.Post(
			"/unsubscribe",
			handler.Unsubscribe,
		)
	})

	// -------------------------------------------------
	// ADMINISTRATOR ENDPOINTS
	// -------------------------------------------------

	router.With(
		AdminAuthMiddleware(
			config.AdminToken,
			config.AdminID,
		),
	).Put(
		"/v1/admin/waitlist/config",
		handler.UpdateConfig,
	)
}

// =====================================================
// STANDALONE HTTP HANDLER
// =====================================================

// NewHTTPHandler creates a standalone HTTP handler.
//
// Use this when the waitlist package is being served
// independently. Do not use it inside app.NewServer
// if CORS is already installed on the application router.
func NewHTTPHandler(
	service *Service,
	tokens *UnsubscribeTokenManager,
	config RouteConfig,
) http.Handler {

	router := chi.NewRouter()

	handler := NewHandler(
		service,
		tokens,
	)

	// Apply CORS once at the router level.
	router.Use(
		CORSMiddleware(config.AllowedOrigins),
	)

	RegisterRoutes(
		router,
		handler,
		config,
	)

	return router
}

package waitlist

import (
	"context"
	"crypto/sha256"
	"crypto/subtle"
	"net"
	"net/http"
	"strconv"
	"strings"
	"sync"
	"time"
)

// =====================================================
// ADMIN AUTHENTICATION
// =====================================================

type adminIDContextKey struct{}

// AdminAuthMiddleware protects administrator endpoints.
//
// The configured bearer token authenticates the request.
// The configured administrator UUID identifies the actor
// for database audit records.
//
// This is a single-administrator authentication mechanism.
// It is not a multi-user identity management system.
func AdminAuthMiddleware(
	expectedToken string,
	adminID string,
) func(http.Handler) http.Handler {

	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(
			func(w http.ResponseWriter, r *http.Request) {

				expectedToken = strings.TrimSpace(expectedToken)
				adminID = strings.TrimSpace(adminID)

				if len(expectedToken) < 32 || !validUUID(adminID) {
					writeError(
						w,
						http.StatusServiceUnavailable,
						"Administrator authentication is not configured.",
					)
					return
				}

				authorization := strings.TrimSpace(
					r.Header.Get("Authorization"),
				)

				parts := strings.Fields(authorization)

				if len(parts) != 2 ||
					!strings.EqualFold(parts[0], "Bearer") {

					writeError(
						w,
						http.StatusUnauthorized,
						"Administrator authentication required.",
					)
					return
				}

				providedToken := parts[1]

				providedHash := sha256.Sum256(
					[]byte(providedToken),
				)

				expectedHash := sha256.Sum256(
					[]byte(expectedToken),
				)

				if subtle.ConstantTimeCompare(
					providedHash[:],
					expectedHash[:],
				) != 1 {

					writeError(
						w,
						http.StatusUnauthorized,
						"Invalid administrator credentials.",
					)
					return
				}

				// Identity comes from server configuration,
				// never from client-supplied request data.
				ctx := context.WithValue(
					r.Context(),
					adminIDContextKey{},
					adminID,
				)

				next.ServeHTTP(
					w,
					r.WithContext(ctx),
				)
			},
		)
	}
}

// =====================================================
// PUBLIC REGISTRATION RATE LIMITER
// =====================================================

type rateLimitEntry struct {
	count       int
	windowStart time.Time
}

type RegistrationRateLimiter struct {
	mu sync.Mutex

	entries map[string]rateLimitEntry

	limit  int
	window time.Duration
	now    func() time.Time

	// Avoid scanning the entire map on every request.
	requestCount uint64
}

func NewRegistrationRateLimiter(
	limit int,
	window time.Duration,
) *RegistrationRateLimiter {

	if limit < 1 {
		limit = 5
	}

	if window <= 0 {
		window = time.Minute
	}

	return &RegistrationRateLimiter{
		entries: make(map[string]rateLimitEntry),
		limit:   limit,
		window:  window,
		now:     time.Now,
	}
}

// clientIP obtains the immediate network peer address.
//
// Forwarded headers are deliberately not trusted here.
// Only use forwarded IP addresses if the application
// is behind a correctly configured trusted proxy.
func clientIP(r *http.Request) string {
	remoteAddr := strings.TrimSpace(r.RemoteAddr)

	if remoteAddr == "" {
		return "unknown"
	}

	host, _, err := net.SplitHostPort(remoteAddr)

	if err == nil && host != "" {
		return host
	}

	// Handles addresses without a port, including IPv6.
	return remoteAddr
}

// Middleware applies a fixed-window registration limit.
//
// The default configured limit is five attempts per IP
// address per minute.
//
// This limiter is process-local. It is not distributed
// across multiple application instances.
func (l *RegistrationRateLimiter) Middleware(
	next http.Handler,
) http.Handler {

	return http.HandlerFunc(
		func(w http.ResponseWriter, r *http.Request) {

			ip := clientIP(r)
			now := l.now()

			l.mu.Lock()

			entry, exists := l.entries[ip]

			if !exists ||
				now.Sub(entry.windowStart) >= l.window {

				entry = rateLimitEntry{
					count:       0,
					windowStart: now,
				}
			}

			if entry.count >= l.limit {

				retryAfter := l.window -
					now.Sub(entry.windowStart)

				if retryAfter < 0 {
					retryAfter = 0
				}

				retrySeconds := int(
					retryAfter.Seconds(),
				) + 1

				w.Header().Set(
					"Retry-After",
					strconv.Itoa(retrySeconds),
				)

				l.entries[ip] = entry

				l.mu.Unlock()

				writeError(
					w,
					http.StatusTooManyRequests,
					"Too many registration attempts. Please try again later.",
				)

				return
			}

			entry.count++

			l.entries[ip] = entry

			l.requestCount++

			// Opportunistically remove expired entries
			// every 256 requests rather than scanning the
			// map on every incoming registration request.
			if l.requestCount%256 == 0 {

				for key, value := range l.entries {

					if now.Sub(value.windowStart) >= l.window {
						delete(l.entries, key)
					}
				}
			}

			l.mu.Unlock()

			next.ServeHTTP(w, r)
		},
	)
}

// =====================================================
// CORS
// =====================================================

// CORSMiddleware permits requests only from explicitly
// configured frontend origins.
//
// Wildcard origins and credentialed cross-origin requests
// are deliberately not enabled.
func CORSMiddleware(
	allowedOrigins []string,
) func(http.Handler) http.Handler {

	allowed := make(map[string]struct{}, len(allowedOrigins))

	for _, origin := range allowedOrigins {

		origin = strings.TrimSpace(origin)

		if origin == "" {
			continue
		}

		allowed[origin] = struct{}{}
	}

	allowedMethods := "GET, POST, PUT, OPTIONS"
	allowedHeaders := "Content-Type, Authorization"

	return func(next http.Handler) http.Handler {

		return http.HandlerFunc(
			func(w http.ResponseWriter, r *http.Request) {

				origin := strings.TrimSpace(
					r.Header.Get("Origin"),
				)

				if origin != "" {

					if _, ok := allowed[origin]; !ok {

						writeError(
							w,
							http.StatusForbidden,
							"Origin not allowed.",
						)

						return
					}

					w.Header().Set(
						"Access-Control-Allow-Origin",
						origin,
					)

					addVaryHeader(w, "Origin")

					w.Header().Set(
						"Access-Control-Allow-Methods",
						allowedMethods,
					)

					w.Header().Set(
						"Access-Control-Allow-Headers",
						allowedHeaders,
					)

					w.Header().Set(
						"Access-Control-Max-Age",
						"600",
					)
				}

				// Handle browser preflight requests.
				if r.Method == http.MethodOptions {

					if origin != "" {

						requestedMethod := strings.TrimSpace(
							r.Header.Get("Access-Control-Request-Method"),
						)

						if requestedMethod != "" &&
							!isAllowedCORSMethod(requestedMethod) {

							writeError(
								w,
								http.StatusForbidden,
								"Requested method is not allowed.",
							)

							return
						}

						requestedHeaders := strings.TrimSpace(
							r.Header.Get("Access-Control-Request-Headers"),
						)

						if requestedHeaders != "" &&
							!areAllowedCORSHeaders(requestedHeaders) {

							writeError(
								w,
								http.StatusForbidden,
								"Requested headers are not allowed.",
							)

							return
						}
					}

					w.WriteHeader(http.StatusNoContent)
					return
				}

				next.ServeHTTP(w, r)
			},
		)
	}
}

// =====================================================
// CORS HELPERS
// =====================================================

func isAllowedCORSMethod(method string) bool {
	switch strings.ToUpper(strings.TrimSpace(method)) {
	case http.MethodGet,
		http.MethodPost,
		http.MethodPut:
		return true

	default:
		return false
	}
}

func areAllowedCORSHeaders(headers string) bool {

	allowed := map[string]struct{}{
		"content-type":  {},
		"authorization": {},
	}

	for _, header := range strings.Split(headers, ",") {

		header = strings.ToLower(
			strings.TrimSpace(header),
		)

		if header == "" {
			continue
		}

		if _, ok := allowed[header]; !ok {
			return false
		}
	}

	return true
}

// addVaryHeader preserves existing Vary values.
func addVaryHeader(
	w http.ResponseWriter,
	value string,
) {

	for _, existing := range w.Header().Values("Vary") {

		for _, item := range strings.Split(existing, ",") {

			if strings.EqualFold(
				strings.TrimSpace(item),
				value,
			) {
				return
			}
		}
	}

	w.Header().Add("Vary", value)
}

package waitlist

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/base64"
	"encoding/json"
	"errors"
	"fmt"
	"strings"
	"time"
)

var ErrInvalidUnsubscribeToken = errors.New("invalid or expired unsubscribe token")

type UnsubscribeChannel string

const (
	UnsubscribeLaunchUpdates UnsubscribeChannel = "launch_updates"
	UnsubscribeMarketing     UnsubscribeChannel = "marketing"
)

type UnsubscribeClaims struct {
	RegistrationID string             `json:"rid"`
	Channel        UnsubscribeChannel `json:"channel"`
	ExpiresAt      int64              `json:"exp"`
}

type UnsubscribeTokenManager struct {
	secret []byte
	clock  func() time.Time
}

func NewUnsubscribeTokenManager(secret string) (*UnsubscribeTokenManager, error) {
	if len(secret) < 32 {
		return nil, errors.New(
			"unsubscribe token secret must contain at least 32 characters",
		)
	}

	return &UnsubscribeTokenManager{
		secret: []byte(secret),
		clock:  time.Now,
	}, nil
}

func (m *UnsubscribeTokenManager) Generate(
	registrationID string,
	channel UnsubscribeChannel,
	expiresAt time.Time,
) (string, error) {

	if strings.TrimSpace(registrationID) == "" {
		return "", ErrInvalidUnsubscribeToken
	}

	if channel != UnsubscribeLaunchUpdates &&
		channel != UnsubscribeMarketing {
		return "", ErrInvalidUnsubscribeToken
	}

	if !expiresAt.After(m.clock()) {
		return "", ErrInvalidUnsubscribeToken
	}

	claims := UnsubscribeClaims{
		RegistrationID: registrationID,
		Channel:        channel,
		ExpiresAt:      expiresAt.Unix(),
	}

	payload, err := json.Marshal(claims)
	if err != nil {
		return "", fmt.Errorf("encode unsubscribe claims: %w", err)
	}

	encodedPayload := base64.RawURLEncoding.EncodeToString(payload)
	signature := m.sign(encodedPayload)

	return encodedPayload + "." +
		base64.RawURLEncoding.EncodeToString(signature), nil
}

func (m *UnsubscribeTokenManager) Verify(
	token string,
) (UnsubscribeClaims, error) {

	parts := strings.Split(token, ".")
	if len(parts) != 2 {
		return UnsubscribeClaims{}, ErrInvalidUnsubscribeToken
	}

	providedSignature, err := base64.RawURLEncoding.DecodeString(parts[1])
	if err != nil {
		return UnsubscribeClaims{}, ErrInvalidUnsubscribeToken
	}

	expectedSignature := m.sign(parts[0])

	if !hmac.Equal(providedSignature, expectedSignature) {
		return UnsubscribeClaims{}, ErrInvalidUnsubscribeToken
	}

	payload, err := base64.RawURLEncoding.DecodeString(parts[0])
	if err != nil {
		return UnsubscribeClaims{}, ErrInvalidUnsubscribeToken
	}

	var claims UnsubscribeClaims

	if err := json.Unmarshal(payload, &claims); err != nil {
		return UnsubscribeClaims{}, ErrInvalidUnsubscribeToken
	}

	if claims.RegistrationID == "" ||
		(claims.Channel != UnsubscribeLaunchUpdates &&
			claims.Channel != UnsubscribeMarketing) ||
		claims.ExpiresAt <= m.clock().Unix() {
		return UnsubscribeClaims{}, ErrInvalidUnsubscribeToken
	}

	return claims, nil
}

func (m *UnsubscribeTokenManager) sign(payload string) []byte {
	mac := hmac.New(sha256.New, m.secret)
	_, _ = mac.Write([]byte(payload))
	return mac.Sum(nil)
}

	auditTable        = "waitlist.waitlist_admin_audit"

const registrationColumns = `

// PostgreSQLDB is implemented by pgxpool.Pool.
type PostgreSQLDB interface {
	Begin(ctx context.Context) (pgx.Tx, error)

	QueryRow(
		ctx context.Context,
		sql string,
		args ...any,
	) pgx.Row

	Exec(
		ctx context.Context,
		sql string,
		args ...any,
	) (pgconn.CommandTag, error)
}

type PostgresRepository struct {
	db PostgreSQLDB
}

func NewPostgresRepository(db PostgreSQLDB) *PostgresRepository {
	return &PostgresRepository{
		db: db,
	}
}

// =====================================================
// UUID GENERATION
// =====================================================

// generateUUID creates an RFC 4122 version 4 UUID.
func generateUUID() (string, error) {
	var id [16]byte

	if _, err := rand.Read(id[:]); err != nil {
		return "", fmt.Errorf("generate UUID: %w", err)
	}

	id[6] = (id[6] & 0x0f) | 0x40
	id[8] = (id[8] & 0x3f) | 0x80

	encoded := hex.EncodeToString(id[:])

	return encoded[0:8] + "-" +
		encoded[8:12] + "-" +
		encoded[12:16] + "-" +
		encoded[16:20] + "-" +
		encoded[20:32], nil
}

// =====================================================
// TRANSACTION CLEANUP
// =====================================================

func rollbackTransaction(tx pgx.Tx) {
	ctx, cancel := context.WithTimeout(
		context.Background(),
		5*time.Second,
	)
	defer cancel()

	// A rollback after a successful commit is harmless.
	_ = tx.Rollback(ctx)
}

// =====================================================
// REGISTRATION
// =====================================================

func (r *PostgresRepository) Register(
	ctx context.Context,
	input RegistrationInput,
) (RegistrationResult, error) {

	if r.db == nil {
		return RegistrationResult{}, errors.New(
			"waitlist database is unavailable",
		)
	}

	email, err := NormalizeEmail(input.Email)
	if err != nil {
		return RegistrationResult{}, err
	}

	tx, err := r.db.Begin(ctx)
	if err != nil {
		return RegistrationResult{}, fmt.Errorf(
			"begin registration transaction: %w",
			err,
		)
	}

	defer rollbackTransaction(tx)

	// 1. Lock the counter row.
	//
	// This serialises new registrations. The counter update
	// and registration insert occur in the same transaction.
	var lastPosition int64

	err = tx.QueryRow(
		ctx,
		`
			SELECT last_position
			FROM `+counterTable+`
			WHERE counter_id = 1
			FOR UPDATE
		`,
	).Scan(&lastPosition)

	if err != nil {
		return RegistrationResult{}, fmt.Errorf(
			"lock waitlist counter: %w",
			err,
		)
	}

	// 2. Check whether this email is already registered.
	//
	// Existing registrations retain their original position
	// and consent preferences. Duplicate submissions do not
	// silently change previously recorded consent.
	existing, err := scanRegistration(
		tx.QueryRow(
			ctx,
			`
				SELECT `+registrationColumns+`
				FROM `+registrationTable+`
				WHERE email = $1
			`,
			email,
		),
	)

	if err == nil {
		var total int64

		err = tx.QueryRow(
			ctx,
			`
				SELECT COUNT(*)
				FROM `+registrationTable+`
			`,
		).Scan(&total)

		if err != nil {
			return RegistrationResult{}, fmt.Errorf(
				"count waitlist registrations: %w",
				err,
			)
		}

		if err := tx.Commit(ctx); err != nil {
			return RegistrationResult{}, fmt.Errorf(
				"commit duplicate registration lookup: %w",
				err,
			)
		}

		return RegistrationResult{
			Registration:    existing,
			IsExisting:      true,
			TotalRegistered: total,
		}, nil
	}

	if !errors.Is(err, pgx.ErrNoRows) {
		return RegistrationResult{}, fmt.Errorf(
			"check existing registration: %w",
			err,
		)
	}

	// 3. Allocate the next permanent position.
	//
	// The counter update rolls back if the subsequent insert
	// or transaction commit fails.
	var nextPosition int64

	err = tx.QueryRow(
		ctx,
		`
			UPDATE `+counterTable+`
			SET last_position = last_position + 1
			WHERE counter_id = 1
			RETURNING last_position
		`,
	).Scan(&nextPosition)

	if err != nil {
		return RegistrationResult{}, fmt.Errorf(
			"allocate waitlist position: %w",
			err,
		)
	}

	// 4. Generate registration ID.
	registrationID, err := generateUUID()
	if err != nil {
		return RegistrationResult{}, err
	}

	// 5. Insert the registration.
	var registration Registration

	err = tx.QueryRow(
		ctx,
		`
			INSERT INTO `+registrationTable+` (
				registration_id,
				email,
				position,
				launch_updates_consent,
				marketing_consent,
				consent_recorded_at
			)
			VALUES ($1, $2, $3, $4, $5, NOW())
			RETURNING `+registrationColumns,
		registrationID,
		email,
		nextPosition,
		input.LaunchUpdatesConsent,
		input.MarketingConsent,
	).Scan(
		&registration.RegistrationID,
		&registration.Email,
		&registration.Position,
		&registration.LaunchUpdatesConsent,
		&registration.MarketingConsent,
		&registration.ConsentRecordedAt,
		&registration.LaunchUpdatesUnsubscribedAt,
		&registration.MarketingUnsubscribedAt,
		&registration.CreatedAt,
	)

	if err != nil {
		return RegistrationResult{}, fmt.Errorf(
			"insert waitlist registration: %w",
			err,
		)
	}

	// 6. Retrieve the genuine registration count.
	var total int64

	err = tx.QueryRow(
		ctx,
		`
			SELECT COUNT(*)
			FROM `+registrationTable+`
		`,
	).Scan(&total)

	if err != nil {
		return RegistrationResult{}, fmt.Errorf(
			"count waitlist registrations: %w",
			err,
		)
	}

	// 7. Commit the counter and registration together.
	if err := tx.Commit(ctx); err != nil {
		return RegistrationResult{}, fmt.Errorf(
			"commit waitlist registration: %w",
			err,
		)
	}

	return RegistrationResult{
		Registration:    registration,
		IsExisting:      false,
		TotalRegistered: total,
	}, nil
}

// =====================================================
// UPDATE WAITLIST CONFIGURATION
// =====================================================

func (r *PostgresRepository) UpdateConfig(
	ctx context.Context,
	input WaitlistConfigInput,
) (WaitlistConfig, error) {

	if r.db == nil {
		return WaitlistConfig{}, errors.New(
			"waitlist database is unavailable",
		)
	}

	timezone := strings.TrimSpace(input.Timezone)

	if timezone == "" || input.LaunchAt.IsZero() {
		return WaitlistConfig{}, ErrInvalidConfig
	}

	if _, err := time.LoadLocation(timezone); err != nil {
		return WaitlistConfig{}, ErrInvalidConfig
	}

	if input.UpdatedBy == nil {
		return WaitlistConfig{}, ErrInvalidConfig
	}

	adminID := strings.TrimSpace(*input.UpdatedBy)

	if !validUUID(adminID) {
		return WaitlistConfig{}, ErrInvalidConfig
	}

	tx, err := r.db.Begin(ctx)
	if err != nil {
		return WaitlistConfig{}, fmt.Errorf(
			"begin configuration update: %w",
			err,
		)
	}

	defer rollbackTransaction(tx)

	// 1. Acquire the shared counter lock.
	//
	// The counter is a permanent singleton row. Using it here
	// serialises configuration updates, including the case
	// where the default configuration does not exist yet.
	var lastPosition int64

	err = tx.QueryRow(
		ctx,
		`
			SELECT last_position
			FROM `+counterTable+`
			WHERE counter_id = 1
			FOR UPDATE
		`,
	).Scan(&lastPosition)

	if err != nil {
		return WaitlistConfig{}, fmt.Errorf(
			"lock waitlist configuration: %w",
			err,
		)
	}

	// 2. Read the previous configuration.
	var previousLaunchAt pgtype.Timestamptz
	var previousTimezone pgtype.Text

	err = tx.QueryRow(
		ctx,
		`
			SELECT launch_at, timezone
			FROM `+configTable+`
			WHERE config_key = 'default'
			FOR UPDATE
		`,
	).Scan(
		&previousLaunchAt,
		&previousTimezone,
	)

	if err != nil && !errors.Is(err, pgx.ErrNoRows) {
		return WaitlistConfig{}, fmt.Errorf(
			"read previous launch configuration: %w",
			err,
		)
	}

	// 3. Generate a new configuration UUID.
	configID, err := generateUUID()
	if err != nil {
		return WaitlistConfig{}, err
	}

	// 4. Insert or update the configuration.
	var config WaitlistConfig

	err = tx.QueryRow(
		ctx,
		`
			INSERT INTO `+configTable+` (
				config_id,
				config_key,
				launch_at,
				timezone,
				updated_at,
				updated_by
			)
			VALUES ($1, 'default', $2, $3, NOW(), $4)
			ON CONFLICT (config_key)
			DO UPDATE SET
				launch_at = EXCLUDED.launch_at,
				timezone = EXCLUDED.timezone,
				updated_at = NOW(),
				updated_by = EXCLUDED.updated_by
			RETURNING
				config_id,
				launch_at,
				timezone,
				updated_at,
				updated_by
		`,
		configID,
		input.LaunchAt,
		timezone,
		adminID,
	).Scan(
		&config.ConfigID,
		&config.LaunchAt,
		&config.Timezone,
		&config.UpdatedAt,
		&config.UpdatedBy,
	)

	if err != nil {
		return WaitlistConfig{}, fmt.Errorf(
			"update waitlist configuration: %w",
			err,
		)
	}

	// 5. Generate an independent audit ID.
	auditID, err := generateUUID()
	if err != nil {
		return WaitlistConfig{}, err
	}

	// Convert nullable previous values into SQL parameters.
	var previousLaunchValue any
	var previousTimezoneValue any

	if previousLaunchAt.Valid {
		previousLaunchValue = previousLaunchAt.Time
	}

	if previousTimezone.Valid {
		previousTimezoneValue = previousTimezone.String
	}

	// 6. Record the administrator's change.
	//
	// Configuration update and audit insertion are part of
	// the same transaction. If audit insertion fails, the
	// configuration change also rolls back.
	_, err = tx.Exec(
		ctx,
		`
			INSERT INTO `+auditTable+` (
				audit_id,
				admin_id,
				action,
				previous_launch_at,
				new_launch_at,
				previous_timezone,
				new_timezone
			)
			VALUES (
				$1,
				$2,
				'UPDATE_LAUNCH_CONFIG',
				$3,
				$4,
				$5,
				$6
			)
		`,
		auditID,
		adminID,
		previousLaunchValue,
		config.LaunchAt,
		previousTimezoneValue,
		config.Timezone,
	)

	if err != nil {
		return WaitlistConfig{}, fmt.Errorf(
			"record configuration audit: %w",
			err,
		)
	}

	// 7. Commit configuration and audit together.
	if err := tx.Commit(ctx); err != nil {
		return WaitlistConfig{}, fmt.Errorf(
			"commit configuration update: %w",
			err,
		)
	}

	return config, nil
}

type rowScanner interface {
	Scan(dest ...any) error
}

func scanRegistration(
	row rowScanner,
) (Registration, error) {

	var registration Registration

	var launchUnsubscribedAt pgtype.Timestamptz
	var marketingUnsubscribedAt pgtype.Timestamptz

	err := row.Scan(
		&registration.RegistrationID,
		&registration.Email,
		&registration.Position,
		&registration.LaunchUpdatesConsent,
		&registration.MarketingConsent,
		&registration.ConsentRecordedAt,
		&launchUnsubscribedAt,
		&marketingUnsubscribedAt,
		&registration.CreatedAt,
	)

	if err != nil {
		return Registration{}, err
	}

	if launchUnsubscribedAt.Valid {
		value := launchUnsubscribedAt.Time
		registration.LaunchUpdatesUnsubscribedAt = &value
	}

	if marketingUnsubscribedAt.Valid {
		value := marketingUnsubscribedAt.Time
		registration.MarketingUnsubscribedAt = &value
	}

	return registration, nil
}

func (r *PostgresRepository) unsubscribeByEmail(
	ctx context.Context,
	email string,
	channel UnsubscribeChannel,
) (Registration, error) {

	if r.db == nil {
		return Registration{}, errors.New(
			"waitlist database is unavailable",
		)
	}

	normalised, err := NormalizeEmail(email)
	if err != nil {
		return Registration{}, err
	}

	var query string

	switch channel {

	case UnsubscribeLaunchUpdates:
		query = `
			UPDATE ` + registrationTable + `
			SET
				launch_updates_consent = FALSE,
				launch_updates_unsubscribed_at =
					COALESCE(launch_updates_unsubscribed_at, NOW())
			WHERE email = $1
			RETURNING ` + registrationColumns

	case UnsubscribeMarketing:
		query = `
			UPDATE ` + registrationTable + `
			SET
				marketing_consent = FALSE,
				marketing_unsubscribed_at =
					COALESCE(marketing_unsubscribed_at, NOW())
			WHERE email = $1
			RETURNING ` + registrationColumns

	default:
		return Registration{}, ErrInvalidUnsubscribeToken
	}

	registration, err := scanRegistration(
		r.db.QueryRow(ctx, query, normalised),
	)

	if errors.Is(err, pgx.ErrNoRows) {
		return Registration{}, ErrRegistrationNotFound
	}

	if err != nil {
		return Registration{}, fmt.Errorf(
			"unsubscribe registration by email: %w",
			err,
		)
	}

	return registration, nil
}

func TestRegistrationRateLimiter_ConcurrentRequests(t *testing.T) {
	const (
		limit       = 5
		concurrency = 20
	)

	limiter := NewRegistrationRateLimiter(
		limit,
		time.Minute,
	)

	var successful int
	var rateLimited int

	var resultMu sync.Mutex
	var waitGroup sync.WaitGroup

	handler := limiter.Middleware(
		http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			w.WriteHeader(http.StatusOK)
		}),
	)

	for i := 0; i < concurrency; i++ {
		waitGroup.Add(1)

		go func() {
			defer waitGroup.Done()

			request := newTestRequest(
				http.MethodPost,
				"/v1/waitlist/register",
			)

			request.RemoteAddr = "192.0.2.50:12345"

			response := newTestResponse(handler, request)

			resultMu.Lock()
			defer resultMu.Unlock()

			switch response.Code {
			case http.StatusOK:
				successful++

			case http.StatusTooManyRequests:
				rateLimited++

			default:
				t.Errorf(
					"unexpected response status: %d",
					response.Code,
				)
			}
		}()
	}

	waitGroup.Wait()

	if successful != limit {
		t.Errorf(
			"expected %d successful requests, got %d",
			limit,
			successful,
		)
	}

	if rateLimited != concurrency-limit {
		t.Errorf(
			"expected %d rate-limited requests, got %d",
			concurrency-limit,
			rateLimited,
		)
	}
}

type DBPinger interface {
	Ping(ctx context.Context) error
}

type Server struct {
	handler http.Handler
	db      DBPinger
}

// NewServer initialises the HTTP application.
//
// It does not create databases, modify database permissions,
// or run database migrations. The supplied PostgreSQL pool
// must already be connected to the dedicated waitlist database.
func NewServer(pool *pgxpool.Pool) (*Server, error) {
	if pool == nil {
		return nil, errors.New("database connection pool is required")
	}

	// --------------------------------------------------
	// 1. Validate administrator configuration
	// --------------------------------------------------

	adminToken := strings.TrimSpace(
		os.Getenv("ADMIN_API_TOKEN"),
	)

	if len(adminToken) < 32 {
		return nil, errors.New(
			"ADMIN_API_TOKEN must contain at least 32 characters",
		)
	}

	adminID := strings.TrimSpace(
		os.Getenv("ADMIN_ID"),
	)

	if !validUUID(adminID) {
		return nil, errors.New(
			"ADMIN_ID must be a valid UUID",
		)
	}

	// --------------------------------------------------
	// 2. Initialise unsubscribe token manager
	// --------------------------------------------------

	unsubscribeSecret := os.Getenv(
		"UNSUBSCRIBE_TOKEN_SECRET",
	)

	tokenManager, err := waitlist.NewUnsubscribeTokenManager(
		unsubscribeSecret,
	)
	if err != nil {
		return nil, err
	}

	// --------------------------------------------------
	// 3. Load CORS configuration
	// --------------------------------------------------

	origins := parseAllowedOrigins(
		os.Getenv("CORS_ALLOWED_ORIGINS"),
	)

	// --------------------------------------------------
	// 4. Initialise application dependencies
	// --------------------------------------------------

	repository := waitlist.NewPostgresRepository(pool)

	service := waitlist.NewService(repository)

	handler := waitlist.NewHandler(
		service,
		tokenManager,
	)

	// --------------------------------------------------
	// 5. Initialise HTTP router
	// --------------------------------------------------

	router := chi.NewRouter()

	// Request identification and panic recovery.
	router.Use(middleware.RequestID)
	router.Use(middleware.Recoverer)

	// Security headers.
	router.Use(securityHeaders)

	// Exact-origin CORS allowlist.
	router.Use(
		waitlist.CORSMiddleware(origins),
	)

	// --------------------------------------------------
	// 6. Health check endpoints
	// --------------------------------------------------

	server := &Server{
		handler: router,
		db:      pool,
	}

	router.Get(
		"/health/live",
		server.liveHandler,
	)

	router.Get(
		"/health/ready",
		server.readyHandler,
	)

	// --------------------------------------------------
	// 7. Register waitlist endpoints
	// --------------------------------------------------

	waitlist.RegisterRoutes(
		router,
		handler,
		waitlist.RouteConfig{
			AdminToken:     adminToken,
			AdminID:        adminID,
			AllowedOrigins: origins,
		},
	)

	return server, nil
}

// Handler exposes the configured HTTP handler.
func (s *Server) Handler() http.Handler {
	return s.handler
}

// --------------------------------------------------
// Health checks
// --------------------------------------------------

// liveHandler confirms that the application process
// is running. It does not check database availability.
func (s *Server) liveHandler(
	w http.ResponseWriter,
	r *http.Request,
) {
	writeJSON(
		w,
		http.StatusOK,
		map[string]interface{}{
			"success": true,
			"status":  "alive",
		},
	)
}

// readyHandler confirms that the application can
// communicate with its PostgreSQL database.
func (s *Server) readyHandler(
	w http.ResponseWriter,
	r *http.Request,
) {
	ctx, cancel := context.WithTimeout(
		r.Context(),
		3*time.Second,
	)
	defer cancel()

	if s.db == nil {
		writeJSON(
			w,
			http.StatusServiceUnavailable,
			map[string]interface{}{
				"success": false,
				"status":  "not_ready",
			},
		)
		return
	}

	if err := s.db.Ping(ctx); err != nil {
		// Do not expose database errors or connection details.
		writeJSON(
			w,
			http.StatusServiceUnavailable,
			map[string]interface{}{
				"success": false,
				"status":  "not_ready",
			},
		)
		return
	}

	writeJSON(
		w,
		http.StatusOK,
		map[string]interface{}{
			"success":  true,
			"status":   "ready",
			"database": "connected",
		},
	)
}

// --------------------------------------------------
// Shared HTTP helpers
// --------------------------------------------------

func writeJSON(
	w http.ResponseWriter,
	status int,
	payload interface{},
) {
	w.Header().Set(
		"Content-Type",
		"application/json",
	)

	w.WriteHeader(status)

	_ = json.NewEncoder(w).Encode(payload)
}

// --------------------------------------------------
// Security middleware
// --------------------------------------------------

func securityHeaders(next http.Handler) http.Handler {
	return http.HandlerFunc(
		func(w http.ResponseWriter, r *http.Request) {
			w.Header().Set(
				"X-Content-Type-Options",
				"nosniff",
			)

			w.Header().Set(
				"X-Frame-Options",
				"DENY",
			)

			w.Header().Set(
				"Referrer-Policy",
				"no-referrer",
			)

			w.Header().Set(
				"Cache-Control",
				"no-store",
			)

			next.ServeHTTP(w, r)
		},
	)
}

// CORS configuration
// --------------------------------------------------

// parseAllowedOrigins converts the comma-separated
// environment variable into a clean origin allowlist.
func parseAllowedOrigins(value string) []string {
	if strings.TrimSpace(value) == "" {
		return nil
	}

	rawOrigins := strings.Split(value, ",")

	origins := make([]string, 0, len(rawOrigins))

	for _, origin := range rawOrigins {
		origin = strings.TrimSpace(origin)

		if origin == "" {
			continue
		}

		origins = append(origins, origin)
	}

	return origins
}

// --------------------------------------------------
// UUID validation
// --------------------------------------------------

func validUUID(value string) bool {
	if len(value) != 36 {
		return false
	}

	for i, character := range value {
		if i == 8 || i == 13 || i == 18 || i == 23 {
			if character != '-' {
				return false
			}
			continue
		}

		isDigit := character >= '0' && character <= '9'
		isLowerHex := character >= 'a' && character <= 'f'
		isUpperHex := character >= 'A' && character <= 'F'

		if !isDigit && !isLowerHex && !isUpperHex {
			return false
		}
	}

	return true
}

// Keep structured application logging available.
//
// Never log:
// - Resident email addresses
// - Unsubscribe tokens
// - Authorization headers
// - Request bodies
// - Sensitive registration information
func applicationLogger() *slog.Logger {
	return slog.New(
		slog.NewJSONHandler(
			os.Stdout,
			nil,
		),
	)
}
